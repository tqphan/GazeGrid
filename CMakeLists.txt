cmake_minimum_required(VERSION 3.24)
project(OptiKeyPro VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Quick Qml DBus Test Svg)
set(TOBII_SDK "/home/q/Downloads/tobii_eye_tracker_linux_installer/stream_engine_linux_3.0.4.6031" CACHE PATH "User-supplied Tobii SDK")
find_package(PkgConfig REQUIRED)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_package(LayerShellQt REQUIRED)
  pkg_check_modules(LIBEI REQUIRED IMPORTED_TARGET libei-1.0)
  pkg_check_modules(SPEECHD REQUIRED IMPORTED_TARGET speech-dispatcher)
  find_package(Qt6 REQUIRED COMPONENTS WaylandClient WaylandClientPrivate)
  pkg_check_modules(WAYLAND REQUIRED IMPORTED_TARGET wayland-client)
  pkg_check_modules(XKBCOMMON REQUIRED IMPORTED_TARGET xkbcommon)
  find_program(WAYLAND_SCANNER wayland-scanner REQUIRED)
  set(IM_PROTOCOL "${CMAKE_CURRENT_SOURCE_DIR}/protocols/input-method-unstable-v1.xml")
  add_custom_command(OUTPUT input-method-client.h COMMAND "${WAYLAND_SCANNER}" client-header "${IM_PROTOCOL}" input-method-client.h DEPENDS "${IM_PROTOCOL}")
  add_custom_command(OUTPUT input-method-protocol.c COMMAND "${WAYLAND_SCANNER}" private-code "${IM_PROTOCOL}" input-method-protocol.c DEPENDS "${IM_PROTOCOL}")
  add_library(optikey_input_method src/PlasmaInputMethod.cpp input-method-protocol.c input-method-client.h)
  target_include_directories(optikey_input_method PUBLIC src PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_link_libraries(optikey_input_method PUBLIC Qt6::Gui PRIVATE Qt6::DBus Qt6::WaylandClientPrivate PkgConfig::WAYLAND PkgConfig::XKBCOMMON optikey_core)
endif()
add_library(optikey_speech src/SpeechOutput.cpp)
target_include_directories(optikey_speech PUBLIC src)
target_link_libraries(optikey_speech PUBLIC Qt6::Core PRIVATE PkgConfig::SPEECHD)
add_library(optikey_core src/KeyboardPack.cpp src/KeyboardLayout.cpp src/KeyNames.cpp src/ActionRunner.cpp src/Dwell.cpp src/ProfileStore.cpp src/PointerActions.cpp src/LookScroll.cpp src/WordPrediction.cpp src/ShortcutMappings.cpp src/PortalPermissionStore.cpp)
target_include_directories(optikey_core PUBLIC src)
target_link_libraries(optikey_core PUBLIC Qt6::Core)
qt_add_executable(optikey-pro src/main.cpp src/TobiiInput.cpp src/PortalInput.cpp src/Controller.cpp src/SymbolProvider.cpp src/ScreenshotPortal.cpp src/PointerCheck.cpp)
set(OPTIKEY_EXECUTABLE "\"${CMAKE_CURRENT_SOURCE_DIR}/run.sh\"")
configure_file(resources/org.optikey.Pro.InputMethod.desktop.in
  org.optikey.Pro.InputMethod.desktop @ONLY)
target_include_directories(optikey-pro PRIVATE "${TOBII_SDK}/include")
target_compile_definitions(optikey-pro PRIVATE TOBII_LIBRARY_PATH="${TOBII_SDK}/lib/x64/libtobii_stream_engine.so" DEFAULT_PACK_PATH="${CMAKE_CURRENT_SOURCE_DIR}/references/optikey-keyboard-en-us")
target_link_libraries(optikey-pro PRIVATE optikey_core optikey_speech Qt6::Gui Qt6::Quick Qt6::Qml Qt6::DBus Qt6::Svg)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
 target_link_libraries(optikey-pro PRIVATE LayerShellQt::Interface PkgConfig::LIBEI optikey_input_method)
 target_compile_definitions(optikey-pro PRIVATE OPTIKEY_LINUX)
endif()
qt_add_resources(optikey-pro ui PREFIX "/" FILES ui/Main.qml ui/InputTarget.qml ui/InputMethodTarget.qml ui/PointerTarget.qml resources/icons.json resources/EnglishUS.dic resources/linux-mappings.json)
enable_testing()
qt_add_executable(core-tests tests/CoreTests.cpp)
target_link_libraries(core-tests PRIVATE optikey_core Qt6::Test)
target_compile_definitions(core-tests PRIVATE PACK_PATH="${CMAKE_CURRENT_SOURCE_DIR}/references/optikey-keyboard-en-us")
qt_add_resources(core-tests mapping_resources PREFIX "/" FILES resources/linux-mappings.json)
add_test(NAME core-tests COMMAND core-tests)
qt_add_executable(controller-tests tests/ControllerTests.cpp src/Controller.cpp src/PortalInput.cpp src/TobiiInput.cpp src/ScreenshotPortal.cpp)
target_include_directories(controller-tests PRIVATE "${TOBII_SDK}/include")
target_compile_definitions(controller-tests PRIVATE TOBII_LIBRARY_PATH="${TOBII_SDK}/lib/x64/libtobii_stream_engine.so" DEFAULT_PACK_PATH="${CMAKE_CURRENT_SOURCE_DIR}/references/optikey-keyboard-en-us")
target_link_libraries(controller-tests PRIVATE optikey_core optikey_speech Qt6::Gui Qt6::DBus Qt6::Test PkgConfig::LIBEI)
qt_add_resources(controller-tests test_resources PREFIX "/" FILES resources/EnglishUS.dic resources/linux-mappings.json)
add_test(NAME controller-tests COMMAND controller-tests)
find_program(DBUS_RUN_SESSION dbus-run-session REQUIRED)
qt_add_executable(screenshot-tests tests/ScreenshotTests.cpp src/ScreenshotPortal.cpp)
target_include_directories(screenshot-tests PRIVATE src)
target_link_libraries(screenshot-tests PRIVATE Qt6::Gui Qt6::DBus Qt6::Test)
add_test(NAME screenshot-tests COMMAND "${DBUS_RUN_SESSION}" -- $<TARGET_FILE:screenshot-tests>)

qt_add_executable(speech-tests tests/SpeechTests.cpp)
target_link_libraries(speech-tests PRIVATE optikey_speech Qt6::Test)
add_test(NAME speech-tests COMMAND speech-tests)
qt_add_executable(portal-tests tests/PortalTests.cpp src/PortalInput.cpp)
target_link_libraries(portal-tests PRIVATE optikey_core Qt6::Gui Qt6::DBus Qt6::Test PkgConfig::LIBEI)
add_test(NAME portal-tests COMMAND "${DBUS_RUN_SESSION}" -- $<TARGET_FILE:portal-tests>)
