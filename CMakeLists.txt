  find_package(Qt6 REQUIRED COMPONENTS WaylandClient WaylandClientPrivate)
  pkg_check_modules(WAYLAND REQUIRED IMPORTED_TARGET wayland-client)
  pkg_check_modules(XKBCOMMON REQUIRED IMPORTED_TARGET xkbcommon)
  find_program(WAYLAND_SCANNER wayland-scanner REQUIRED)
  set(IM_PROTOCOL "${CMAKE_CURRENT_SOURCE_DIR}/protocols/input-method-unstable-v1.xml")
  add_custom_command(OUTPUT input-method-client.h COMMAND "${WAYLAND_SCANNER}" client-header "${IM_PROTOCOL}" input-method-client.h DEPENDS "${IM_PROTOCOL}")
  add_custom_command(OUTPUT input-method-protocol.c COMMAND "${WAYLAND_SCANNER}" private-code "${IM_PROTOCOL}" input-method-protocol.c DEPENDS "${IM_PROTOCOL}")
  add_library(gazegrid_input_method src/PlasmaInputMethod.cpp input-method-protocol.c input-method-client.h)
  target_include_directories(gazegrid_input_method PUBLIC src PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_link_libraries(gazegrid_input_method PUBLIC Qt6::Gui PRIVATE LayerShellQt::Interface Qt6::DBus Qt6::WaylandClientPrivate PkgConfig::WAYLAND PkgConfig::XKBCOMMON gazegrid_core)
