cmake_minimum_required(VERSION 3.24)
project(gaze-gridWindowsFoundation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Test)
add_library(gaze-grid_windows_input ../../src/WindowsInput.cpp ../../src/KeyNames.cpp)
target_include_directories(gaze-grid_windows_input PUBLIC ../../src)
target_link_libraries(gaze-grid_windows_input PUBLIC Qt6::Core)
if(WIN32)
    target_link_libraries(gaze-grid_windows_input PRIVATE user32)
endif()
enable_testing()
add_executable(windows-input-tests ../../tests/WindowsInputTests.cpp)
target_link_libraries(windows-input-tests PRIVATE gaze-grid_windows_input Qt6::Test)
add_test(NAME windows-input-tests COMMAND windows-input-tests)
